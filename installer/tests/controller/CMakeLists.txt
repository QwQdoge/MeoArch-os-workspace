cmake_minimum_required(VERSION 3.21)
project(MeoInstallerControllerTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.6 REQUIRED COMPONENTS Core Network Qml Test)
enable_testing()
add_executable(test_plan_state test_plan_state.cpp ../../app/installercontroller.cpp ../../app/installercontroller.h
    ../../app/terminalsession.cpp ../../app/terminalsession.h)
target_link_libraries(test_plan_state PRIVATE Qt6::Core Qt6::Network Qt6::Qml Qt6::Test)
if(UNIX AND NOT APPLE)
    target_link_libraries(test_plan_state PRIVATE util)
endif()
add_test(NAME installer_plan_state COMMAND test_plan_state)
